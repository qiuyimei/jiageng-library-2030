24export function evaluateLegacyCandidate(input, clockSeconds = 32400, cooldownUntil = 0) {
25 const x = validateRuleInput(input);

48export function pilotReducer(state, event) {
49 const next = structuredClone(state);
50 if (event.type === 'conditions') {
51 for (const key of ['objection', 'reserved']) if (key in event) {
52 if (typeof event[key] !== 'boolean') throw new Error('条件须为布尔值');
53 next[key] = event[key];
54 }
55 next.version++;
56 next.pending = null;
57 if (next.mode === 'active' && (next.objection || next.reserved)) next.manualReview = true;
58 return {state: next, result: next.manualReview ? 'manual-review' : 'conditions-updated'};
59 }
60 if (event.type === 'request') {
61 const failed = blockers(next);
62 if (next.mode === 'active') return {state: next, result: 'already-active'};
63 if (failed.length) return {state: next, result: 'blocked', failed};
64 next.pending = {version: next.version, floor: next.floor};
65 return {state: next, result: 'pending'};
66 }
67 if (event.type === 'approve') {
68 if (!next.pending) return {state: next, result: 'no-pending'};
69 const failed = blockers(next);
70 if (event.expectedVersion !== next.version || next.pending.version !== next.version) {
71 next.pending = null;
72 return {state: next, result: 'stale-approval'};
73 }
74 next.pending = null;
75 if (failed.length) return {state: next, result: 'blocked', failed};
76 next.mode = 'active';
77 return {state: next, result: next.floor === 5 ? 'quiet-prompts-only' : 'pilot-activated'};
78 }
79 if (event.type === 'restore') {
80 if (next.manualReview) return {state: next, result: 'manual-review'};
81 if (next.mode !== 'active') return {state: next, result: 'not-active'};
82 if (!Number.isFinite(event.vacancySec) || event.vacancySec < 300) return {state: next, result: 'wait-for-continuous-vacancy'};
83 // 服务状态复原与实体设施复原分开确认；二者都满足才完成教学复原。
84 if (event.staffConfirmed !== true) return {state: next, result: 'await-staff-confirmation'};
85 next.mode = 'normal';
86 next.cooldown = true;
87 next.version++;
88 return {state: next, result: 'restored'};
89 }
90 if (event.type === 'advance') {
91 if (!Number.isFinite(event.seconds) || event.seconds < 600) return {state: next, result: 'cooldown-not-finished'};
92 next.cooldown = false;
93 return {state: next, result: 'cooldown-finished'};
94 }
95 if (event.type === 'staff-clear') {
96 if (next.objection || next.reserved) return {state: next, result: 'constraints-still-present'};
97 next.manualReview = false;
98 return {state: next, result: 'staff-cleared'};
99 }
100 throw new Error('未知事件');
101}
